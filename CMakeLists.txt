cmake_minimum_required(VERSION 3.20)
project(t27 VERSION 0.2.0 DESCRIPTION "T27 balanced-ternary core (review candidate)" LANGUAGES CXX)
include(CTest)
option(T27_BUILD_TESTS "Build tests" ${BUILD_TESTING})
option(T27_BUILD_DOCS "Build Doxygen documentation" OFF)
option(T27_BUILD_BENCH "Build benchmark" OFF)
option(T27_INSTALL "Enable installation and CPack" ON)
option(T27_WARNINGS_AS_ERRORS "Treat warnings as errors" OFF)
option(T27_UNITY_BUILD "Use unity compilation" OFF)
option(T27_ENABLE_SANITIZERS "Enable address/undefined sanitizers" OFF)
option(T27_ENABLE_COVERAGE "Enable gcov-compatible coverage" OFF)
option(T27_ENABLE_LLVM_COVERAGE "Enable LLVM source coverage" OFF)
option(T27_BUILD_EXPERIMENTAL_CPU "Build the experimental ISA v0.1 emulator (not installed)" OFF)
option(T27_BUILD_FUZZ "Build Clang libFuzzer target" OFF)
# Accepted for old callers; there is now one fallback-free algorithm in all modes.
option(T27_LONGDIV_STRICT "Compatibility option (division is always strict)" ON)
option(T27_LONGDIV_NO_FALLBACK "Compatibility option (no fallback exists)" ON)
if(T27_ENABLE_COVERAGE AND T27_ENABLE_LLVM_COVERAGE)
  message(FATAL_ERROR "Choose one coverage format")
endif()
include(cmake/CompilerWarnings.cmake)
include(cmake/Sanitizers.cmake)
include(cmake/Coverage.cmake)
function(t27_configure_target target)
  target_compile_features(${target} PUBLIC cxx_std_20)
  t27_setup_warnings(${target})
  if(T27_ENABLE_SANITIZERS)
    t27_enable_sanitizers(${target})
  endif()
  if(T27_ENABLE_COVERAGE OR T27_ENABLE_LLVM_COVERAGE)
    t27_enable_coverage(${target})
  endif()
endfunction()
set(T27_SOURCES src/t27-core/convert.cpp src/t27-core/ops.cpp src/t27-core/div.cpp src/t27-core/div_round.cpp)
add_library(t27 STATIC ${T27_SOURCES})
add_library(T27::t27 ALIAS t27)
target_include_directories(t27 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
t27_configure_target(t27)
set_target_properties(t27 PROPERTIES UNITY_BUILD ${T27_UNITY_BUILD})
if(T27_BUILD_EXPERIMENTAL_CPU)
  add_library(t27_cpu STATIC experimental/cpu/src/isa.cpp experimental/cpu/src/cpu.cpp experimental/cpu/src/assembler.cpp)
  target_include_directories(t27_cpu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/experimental/cpu/include)
  target_link_libraries(t27_cpu PUBLIC T27::t27)
  t27_configure_target(t27_cpu)
  add_executable(t27_run experimental/cpu/tools/run.cpp)
  target_link_libraries(t27_run PRIVATE t27_cpu)
  t27_configure_target(t27_run)
  add_executable(t27_cpu_demo experimental/cpu/examples/demo.cpp)
  target_link_libraries(t27_cpu_demo PRIVATE t27_cpu)
  t27_configure_target(t27_cpu_demo)
endif()
if(T27_BUILD_TESTS)
  enable_testing()
  if(T27_BUILD_EXPERIMENTAL_CPU)
    add_executable(t27_cpu_trace_bridge experimental/cpu/tests/cpu_trace_bridge.cpp)
    target_link_libraries(t27_cpu_trace_bridge PRIVATE t27_cpu)
    t27_configure_target(t27_cpu_trace_bridge)
    add_executable(t27_cpu_tests experimental/cpu/tests/test_cpu.cpp)
    target_link_libraries(t27_cpu_tests PRIVATE t27_cpu)
    target_include_directories(t27_cpu_tests PRIVATE tests)
    t27_configure_target(t27_cpu_tests)
    add_test(NAME t27.cpu COMMAND t27_cpu_tests)
    add_test(NAME t27.cpu_demo COMMAND t27_cpu_demo)
    add_executable(t27_call_tests experimental/cpu/tests/test_calls.cpp)
    target_link_libraries(t27_call_tests PRIVATE t27_cpu)
    target_include_directories(t27_call_tests PRIVATE tests)
    t27_configure_target(t27_call_tests)
    add_test(NAME t27.calls COMMAND t27_call_tests)
    set_tests_properties(t27.calls PROPERTIES TIMEOUT 30)
    add_executable(t27_assembler_tests experimental/cpu/tests/test_assembler.cpp)
    target_link_libraries(t27_assembler_tests PRIVATE t27_cpu)
    target_include_directories(t27_assembler_tests PRIVATE tests)
    t27_configure_target(t27_assembler_tests)
    add_test(NAME t27.assembler COMMAND t27_assembler_tests)
    set(T27_CPU_HEADER_UNITS "")
    foreach(header IN ITEMS isa cpu assembler)
      set(unit "${CMAKE_CURRENT_BINARY_DIR}/header-tests/experimental_${header}.cpp")
      file(WRITE "${unit}" "#include <t27/experimental/${header}.hpp>\n")
      list(APPEND T27_CPU_HEADER_UNITS "${unit}")
    endforeach()
    add_library(t27_cpu_header_checks OBJECT ${T27_CPU_HEADER_UNITS})
    target_link_libraries(t27_cpu_header_checks PRIVATE t27_cpu)
    t27_configure_target(t27_cpu_header_checks)
  endif()
  add_executable(t27_tests tests/test_main.cpp tests/test_fixed.cpp tests/test_fixed_shifts.cpp tests/test_div3k.cpp tests/test_divmod.cpp tests/test_div_round.cpp)
  add_executable(t27_contract tests/test_contract.cpp)
  add_executable(t27_extra tests/test_div_extra.cpp)
  add_executable(t27_oracle_bridge tests/oracle_bridge.cpp)
  add_executable(t27_example examples/basic.cpp)
  add_executable(t27_fuzz_smoke tests/fuzz_smoke.cpp fuzz/t27_fuzz_div.cpp)
  foreach(target IN ITEMS t27_tests t27_contract t27_extra t27_oracle_bridge t27_example t27_fuzz_smoke)
    target_link_libraries(${target} PRIVATE T27::t27)
    target_include_directories(${target} PRIVATE tests)
    t27_configure_target(${target})
  endforeach()
  add_test(NAME t27.basic COMMAND t27_tests)
  add_test(NAME t27.contract COMMAND t27_contract)
  add_test(NAME t27.extra COMMAND t27_extra)
  add_test(NAME t27.example COMMAND t27_example)
  add_test(NAME t27.fuzz_smoke COMMAND t27_fuzz_smoke)
  find_package(Python3 3.9 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    if(T27_BUILD_EXPERIMENTAL_CPU)
      add_test(NAME t27.cpu_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/experimental/cpu/tests/cpu_oracle.py $<TARGET_FILE:t27_cpu_trace_bridge>)
      set_tests_properties(t27.cpu_oracle PROPERTIES TIMEOUT 180)
      add_test(NAME t27.assembler_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/experimental/cpu/tests/test_cli.py $<TARGET_FILE:t27_run> ${CMAKE_CURRENT_SOURCE_DIR}/experimental/cpu/programs)
      set_tests_properties(t27.assembler_cli PROPERTIES TIMEOUT 60)
    endif()
    add_test(NAME t27.integer_oracle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/oracle.py $<TARGET_FILE:t27_oracle_bridge>)
    set_tests_properties(t27.integer_oracle PROPERTIES TIMEOUT 180)
  else()
    message(WARNING "Python 3.9+ missing: independent integer oracle will NOT run")
  endif()
  # Compile every public header in isolation to catch missing includes.
  file(GLOB_RECURSE T27_HEADERS CONFIGURE_DEPENDS RELATIVE ${CMAKE_CURRENT_SOURCE_DIR}/include include/*.hpp)
  set(T27_HEADER_UNITS "")
  foreach(header IN LISTS T27_HEADERS)
    string(MAKE_C_IDENTIFIER "${header}" stem)
    set(unit "${CMAKE_CURRENT_BINARY_DIR}/header-tests/${stem}.cpp")
    file(WRITE "${unit}" "#include <${header}>\n")
    list(APPEND T27_HEADER_UNITS "${unit}")
  endforeach()
  add_library(t27_header_checks OBJECT ${T27_HEADER_UNITS})
  target_link_libraries(t27_header_checks PRIVATE T27::t27)
  t27_configure_target(t27_header_checks)
  include(cmake/CoverageTargets.cmake)
endif()

if(T27_BUILD_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR MSVC)
    message(FATAL_ERROR "T27_BUILD_FUZZ requires Clang with libFuzzer (not clang-cl)")
  endif()
  add_library(t27_fuzz_support STATIC ${T27_SOURCES})
  target_include_directories(t27_fuzz_support PUBLIC include)
  target_compile_features(t27_fuzz_support PUBLIC cxx_std_20)
  target_compile_options(t27_fuzz_support PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
  add_executable(t27_fuzz_div fuzz/t27_fuzz_div.cpp)
  target_link_libraries(t27_fuzz_div PRIVATE t27_fuzz_support)
  target_compile_options(t27_fuzz_div PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
  target_link_options(t27_fuzz_div PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
